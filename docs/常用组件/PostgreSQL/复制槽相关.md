
```sql 
-- 查询当前lsn
SELECT pg_current_wal_lsn();
select * from pg_current_xlog_location();


-- pgoutput 必须使用 publication
-- 检查是否存在 publication
SELECT pubname FROM pg_publication;

-- 查看发布的表
SELECT * FROM pg_publication_tables WHERE pubname = 'test_pub';

-- 创建 publication
CREATE PUBLICATION "test_pub";
CREATE PUBLICATION "test_pub" FOR TABLE t1, t2;
CREATE PUBLICATION "test_pub_alltables" FOR ALL TABLES;
       
-- 修改 publication
ALTER PUBLICATION "test_pub" ADD TABLE t1,t2,t3;
ALTER PUBLICATION "test_pub" DROP TABLE t1,t2;
ALTER PUBLICATION "test_pub" SET TABLE t2,t3;
       
-- 删除 publication
DROP PUBLICATION IF EXISTS "test_pub";
DROP PUBLICATION IF EXISTS "test_pub_alltables";

       
       
-- 创建复制槽
select * from pg_create_logical_replication_slot('test_wal2json', 'wal2json');
select * from pg_create_logical_replication_slot('test_pgoutput', 'pgoutput');

-- 查询复制槽
select * from pg_replication_slots;

-- 删除复制槽
select pg_drop_replication_slot('test_wal2json');
select pg_drop_replication_slot('test_pgoutput');

-- 查看复制槽数据 wal2json
SELECT * FROM pg_logical_slot_peek_changes('test_wal2json', NULL, 4096);
-- 查看复制槽数据 pgoutput
SELECT * FROM pg_logical_slot_peek_binary_changes( 'test_pgoutput', NULL, 4096, 'proto_version',
                                                   '1', 'publication_names', 'test_pub' );
```
